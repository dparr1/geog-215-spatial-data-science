% PrettyPDF v0.0.5 (nrennie/PrettyPDF), changed for this book: the heading
% styles use KOMA-Script's own font commands (Quarto books use the KOMA
% class scrreprt) instead of the sectsty and titlesec packages. Those two
% replace KOMA's heading code, and with recent LaTeX versions that printed
% a "0" (the chapter number before Chapter 1) in front of unnumbered
% chapters such as the Table of contents and the Preface, and broke unit
% (part) pages. Don't load titlesec or sectsty again in another header file.
% load packages
\usepackage{geometry}
\usepackage{xcolor}
\usepackage{eso-pic}
\usepackage{fancyhdr}
\usepackage{fontspec}

%% Set page size with a wider right margin
\geometry{a4paper, total={170mm,257mm}, left=20mm, top=20mm, bottom=20mm, right=50mm}

%% Let's define some colours
\definecolor{light}{HTML}{E6E6FA}
\definecolor{highlight}{HTML}{800080}
\definecolor{dark}{HTML}{330033}

%% Let's add the border on the right hand side 
\AddToShipoutPicture{% 
    \AtPageLowerLeft{% 
        \put(\LenToUnit{\dimexpr\paperwidth-3cm},0){% 
            \color{light}\rule{3cm}{\LenToUnit\paperheight}%
          }%
     }%
     % logo
    \AtPageLowerLeft{% start the bar at the bottom right of the page
        \put(\LenToUnit{\dimexpr\paperwidth-2.25cm},27.2cm){% move it to the top right
            \color{light}\includegraphics[width=1.5cm]{_extensions/nrennie/PrettyPDF/logo.png}
          }%
     }%
}

%% Style the page number
\fancypagestyle{mystyle}{
  \fancyhf{}
  \renewcommand\headrulewidth{0pt}
  \fancyfoot[R]{\thepage}
  \fancyfootoffset{3.5cm}
}
\setlength{\footskip}{20pt}

%% style the chapter/section fonts (KOMA-Script; see the note at the top)
\addtokomafont{disposition}{\color{dark}}
\setkomafont{chapter}{\fontsize{20}{24}\selectfont}
\setkomafont{section}{\fontsize{20}{24}\selectfont}
\setkomafont{subsection}{\Large}
%% a rule under subsection headings
\makeatletter
\renewcommand{\sectionlinesformat}[4]{%
  \Ifstr{#1}{subsection}{%
    \parbox[t]{\linewidth}{\raggedsection\@hangfrom{\hskip #2#3}{#4}\par
      \kern 3pt\hrule height 0.8pt}%
  }{\@hangfrom{\hskip #2#3}{#4}}%
}
\makeatother

% left align title
\makeatletter
\renewcommand{\maketitle}{\bgroup\setlength{\parindent}{0pt}
\begin{flushleft}
  {\sffamily\huge\textbf{\MakeUppercase{\@title}}} \vspace{0.3cm} \newline
  {\Large {\@subtitle}} \newline
  \@author
\end{flushleft}\egroup
}
\makeatother

%% Fonts (GEOG 215 change): Atkinson Hyperlegible Next for text and headings
%% and Atkinson Hyperlegible Mono for code, to match the website
%% (assets/fonts.scss). Static TTFs from Google Fonts, SIL Open Font License,
%% in assets/pdf-fonts/atkinson/. (PrettyPDF's own Ubuntu files are unused.)
\setmainfont{AtkinsonHyperlegibleNext}[
    Path=assets/pdf-fonts/atkinson/,

    Extension = .ttf,
    UprightFont=*-Regular,
    BoldFont=*-Bold,
    ItalicFont=*-Italic,
    BoldItalicFont=*-BoldItalic,
    ]

\setsansfont{AtkinsonHyperlegibleNext}[
    Path=assets/pdf-fonts/atkinson/,
    Scale=0.9,
    Extension = .ttf,
    UprightFont=*-Regular,
    BoldFont=*-Bold,
    ItalicFont=*-Italic,
    BoldItalicFont=*-BoldItalic,
    ]

\setmonofont{AtkinsonHyperlegibleMono}[
    Path=assets/pdf-fonts/atkinson/,
    Scale=MatchLowercase,
    Extension = .ttf,
    UprightFont=*-Regular,
    BoldFont=*-Bold,
    ItalicFont=*-Italic,
    BoldItalicFont=*-BoldItalic,
    ]
